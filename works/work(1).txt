public class Main {
    public static void main(String[] args) {
        int[] numbers = {4, 8, 2, 6, 10};
        int sum = 0;
        int max = numbers[0];
        for (int i = 0; i < numbers.length; i++) {
            sum = sum + numbers[i];
            if (numbers[i] > max) {
                max = numbers[i];
            }
        }
        System.out.println("Сумма: " + sum);
        System.out.println("Максимум: " + max);
    }
}
