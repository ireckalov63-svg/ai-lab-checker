public class Main {
    public static void main(String[] args) {
        int[] x = {4, 8, 2, 6, 10};
        int s = x[0] + x[1] + x[2] + x[3] + x[4];
        int m = x[0];
        if (x[1] > m) m = x[1];
        if (x[2] > m) m = x[2];
        if (x[3] > m) m = x[3];
        if (x[4] > m) m = x[4];
        System.out.println("Сумма: " + s);
        System.out.println("Максимум: " + m);
    }
}
